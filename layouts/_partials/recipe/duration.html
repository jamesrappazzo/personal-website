{{- /* A recipe time as written in front matter ("20 min", "1 h 30 min", "1 hour", "90") → ISO 8601 ("PT1H30M"), or "" when it can't be read (e.g. "overnight"): the page still shows the text, only the JSON-LD skips it. Returns (dict "iso" "PT…" "minutes" ). */ -}} {{- $s := lower (trim (string .) " ") }} {{- $d := 0 }}{{- $h := 0 }}{{- $m := 0 }} {{- with findRESubmatch `(\d+)\s*d` $s 1 }}{{ $d = int (index (index . 0) 1) }}{{ end }} {{- with findRESubmatch `(\d+)\s*h` $s 1 }}{{ $h = int (index (index . 0) 1) }}{{ end }} {{- with findRESubmatch `(\d+)\s*m` $s 1 }}{{ $m = int (index (index . 0) 1) }}{{ end }} {{- if findRE `^\d+$` $s }}{{ $m = int $s }}{{ end }} {{- $minutes := add (mul $d 1440) (add (mul $h 60) $m) }} {{- $iso := "" }} {{- if gt $minutes 0 }} {{- $iso = "PT" }} {{- $hours := div $minutes 60 }} {{- with $hours }}{{ $iso = printf "%s%dH" $iso . }}{{ end }} {{- with mod $minutes 60 }}{{ $iso = printf "%s%dM" $iso . }}{{ end }} {{- end }} {{- return (dict "iso" $iso "minutes" $minutes) }}
