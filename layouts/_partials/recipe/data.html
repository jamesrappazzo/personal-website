{{- /* Structured recipe data read from a recipe page, for the page itself and its schema.org JSON-LD. The Markdown is the source of truth: `## Ingredients` (list items, optionally grouped under `###` headings) and `## Instructions` (or Method / Directions / Steps: one list item per step, `###` headings make sections). A section heading may carry text after its keyword ("## Ingredients (serves 2)"); fenced code blocks are ignored. Times, yield and the rest come from front matter. */ -}} {{- /* The Markdown without fenced code blocks (``` or ~~~, closed by a fence of the same character at least as long). */}} {{- $raw := "" }}{{- $fence := "" }} {{- range split .RawContent "\n" }} {{- $line := . }} {{- if $fence }} {{- with findRESubmatch "^[ \\t]{0,3}(`{3,}|~{3,})[ \\t]*$" $line 1 }} {{- $f := index (index . 0) 1 }} {{- if and (eq (substr $f 0 1) (substr $fence 0 1)) (ge (len $f) (len $fence)) }}{{ $fence = "" }}{{ end }} {{- end }} {{- $line = "" }} {{- else }} {{- with findRESubmatch "^[ \\t]{0,3}(`{3,}|~{3,})" $line 1 }}{{ $fence = index (index . 0) 1 }}{{ $line = "" }}{{ end }} {{- end }} {{- $raw = printf "%s%s\n" $raw $line }} {{- end }} {{- $clean := `(?m)^[ \t]*(?:[-*+]|\d+[.)])[ \t]+(?:\[[ xX]\][ \t]+)?` }} {{- /* Ingredients: a flat list (for the JSON-LD) and the same items grouped under their ### headings (for Copy). */}} {{- $ingredients := slice }}{{- $groups := slice }} {{- with findRESubmatch `(?ims)^##[ \t]+ingredients\b[^\n]*$(.*?)(?:^##[ \t]|\z)` $raw 1 }} {{- $name := "" }}{{- $items := slice }} {{- range split (index (index . 0) 1) "\n" }} {{- with findRESubmatch `^###[ \t]+(.+?)[ \t#]*$` . 1 }} {{- if $items }}{{ $groups = $groups | append (dict "name" $name "items" $items) }}{{ end }} {{- $name = index (index . 0) 1 | markdownify | plainify | htmlUnescape | strings.TrimSpace }}{{- $items = slice }} {{- else }} {{- if findRE `^[ \t]*(?:[-*+]|\d+[.)])[ \t]+.+$` . }} {{- $item := replaceRE $clean "" . | markdownify | plainify | htmlUnescape | strings.TrimSpace }} {{- $ingredients = $ingredients | append $item }}{{- $items = $items | append $item }} {{- end }} {{- end }} {{- end }} {{- if $items }}{{ $groups = $groups | append (dict "name" $name "items" $items) }}{{ end }} {{- end }} {{- /* Steps: a flat list, or HowToSections when the steps sit under ### headings. */}} {{- $sections := slice }} {{- with findRESubmatch `(?ims)^##[ \t]+(?:instructions|method|directions|steps)\b[^\n]*$(.*?)(?:^##[ \t]|\z)` $raw 1 }} {{- $name := "" }}{{- $steps := slice }} {{- range split (index (index . 0) 1) "\n" }} {{- with findRESubmatch `^###[ \t]+(.+?)[ \t#]*$` . 1 }} {{- if $steps }}{{ $sections = $sections | append (dict "name" $name "steps" $steps) }}{{ end }} {{- $name = index (index . 0) 1 | markdownify | plainify | htmlUnescape }}{{- $steps = slice }} {{- else }} {{- if findRE `^[ \t]*(?:[-*+]|\d+[.)])[ \t]+\S` . }} {{- $steps = $steps | append (replaceRE $clean "" . | markdownify | plainify | htmlUnescape | strings.TrimSpace) }} {{- end }} {{- end }} {{- end }} {{- if $steps }}{{ $sections = $sections | append (dict "name" $name "steps" $steps) }}{{ end }} {{- end }} {{- /* The photo: front matter `image`, else the first image in the text, else any image in the bundle. */}} {{- $image := "" }} {{- with .Params.image }}{{ $image = $.Resources.GetMatch (path.Base .) }}{{ end }} {{- if not $image }} {{- /* The destination is either (may contain spaces) or runs to the first space or ")". */}} {{- with findRESubmatch `!\[[^\]]*\]\((?:<([^>\n]+)>|([^)\s]+))` $raw 1 }} {{- $m := index . 0 }} {{- $image = $.Resources.Get (urls.Parse (index $m 1 | default (index $m 2))).Path }} {{- end }} {{- end }} {{- if not $image }}{{ $image = index (.Resources.ByType "image") 0 }}{{ end }} {{- $prep := partial "recipe/duration.html" (.Params.prep | default "") }} {{- $cook := partial "recipe/duration.html" (.Params.cook | default "") }} {{- $total := partial "recipe/duration.html" (.Params.total | default "") }} {{- if and (not .Params.total) $prep.iso $cook.iso }} {{- $total = partial "recipe/duration.html" (printf "%dm" (add $prep.minutes $cook.minutes)) }} {{- end }} {{- /* The id Hugo gave the ingredients heading, for the jump link (it follows any text after the keyword). */}} {{- $anchor := "" }} {{- if $ingredients }} {{- with findRESubmatch `(?i)]*\sid="([^"]+)"[^>]*>\s*ingredients\b` .Content 1 }}{{ $anchor = index (index . 0) 1 }}{{ end }} {{- end }} {{- return (dict "ingredients" $ingredients "ingredientGroups" $groups "anchor" $anchor "sections" $sections "image" $image "prep" $prep.iso "cook" $cook.iso "total" $total.iso ) }}
